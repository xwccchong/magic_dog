    load_mode_config() 与 ModeCortexRuntime 详解¶

    run.py 启动链中最关键的两块：一个负责把配置文件变成 Python 对象，一个负责用这些对象驱动整个系统运行。

    raw_config = load_env_vars(raw_config)

    # ====== 第5步: Schema 校验 ======
    validate_config_schema(raw_config)
    # 用 config/schema/multi_mode_schema.json 或 single_mode_schema.json
    # 做 JSON Schema 校验 → 必填字段、类型错误等

    # ====== 第6步: 单模式 → 多模式转换 ======
    raw_config = convert_to_multi_mode(raw_config)
    # 如果配置没有 "modes" 顶层 key (如 unitree_go2_koala_welcome.json5)

    关键设计：_raw_* 前缀的含义¶

    @dataclass
class ModeConfig:
    # 实例化后的真实对象
    agent_inputs: List[Sensor] = field(default_factory=list)      # ← 最终存 Sensor 实例
    cortex_llm: Optional[LLM] = None                               # ← 最终存 LLM 实例
    agent_actions: List[AgentAction] = field(default_factory=list) # ← 最终存 Action 实例
    backgrounds: List[Background] = field(default_factory=list)    # ← 最终存 Background 实例

    # 实例化前的原始 dict（由 load_mode_config 填充）
    _raw_inputs: List[Dict] = field(default_factory=list)          # ← 此时还是 dict
    _raw_llm: Optional[Dict] = None                                # ← 此时还是 dict
    _raw_actions: List[Dict] = field(default_factory=list)         # ← 此时还是 dict
    _raw_backgrounds: List[Dict] = field(default_factory=list)     # ← 此时还是 dict

    load_mode_config() 只填充 _raw_* 字段（原始 dict），不填充实例字段。真正的实例化延迟到 _load_mode_components() 中执行（在 _initialize_mode() 时调用）。

    这样设计的原因： - 配置验证和组件加载解耦 - lifecycle hooks 可以在组件加载前先执行 - mode 切换时可以重新加载（销毁旧实例 → 创建新实例）

    __init__() 做了什么¶

    def __init__(self, mode_config, mode_config_name, hot_reload, check_interval):
    # ====== 第1层: 配置持有 ======
    self.mode_config = mode_config            # ModeSystemConfig 对象
    self.mode_config_name = mode_config_name  # "unitree_go2_koala_welcome"

    # ====== 第2层: 核心单例创建 ======
    self.mode_manager = ModeManager(mode_config)
    # 内部:
    #   1. 确定初始模式 (default_mode 或从 memory 恢复)
    #   2. 打开 Zenoh 客户端，订阅 mode/request、mode/context
    #   3. 声明 mode/response publisher
    #   4. 生成 config/memory/.runtime.json5

    self.io_provider = IOProvider()
    # 全局单例:
    #   存储每个 input 的最新值 (按 "Voice"/"Vision"/"GPS" 等 key 索引)
    #   tick 计数器
    #   模式切换输入缓冲区
    #   prompt 历史
    #   fuser 性能计时

    self.sleep_ticker_provider = SleepTickerProvider()
    # tick 间休眠控制 → skip_sleep=True 时跳过休眠 (mode 切换时用)

    self.config_provider = ConfigProvider()
    # HTTP 配置服务器 (可选)

    # ====== 第3层: 所有 Orchestrator 和 Task 初始为空 ======
    self.current_config = None         # RuntimeConfig (由 _initialize_mode 设置)
    self.fuser = None                  # Fuser (由 _initialize_mode 创建)
    self.action_orchestrator = None    # ActionOrchestrator (由 _initialize_mode 创建)
    self.input_orchestrator = None     # (由 _start_orchestrators 创建)
    self.background_orchestrator = None
    self.simulator_orchestrator = None
    self.mcp_orchestrator = None

    self.cortex_loop_task = None       # 6 个 asyncio Task
    self.input_listener_task = None
    self.action_task = None
    self.background_task = None
    self.simulator_task = None
    self.mode_transition_task = None

    # ====== 第4层: 模式切换基础设施 ======
    self.mode_manager.add_transition_callback(self._on_mode_transition)
    # 注册回调: 当 ModeManager 切换 mode 时 → _on_mode_transition 被调用

    self._cortex_loop_generation = 0
    # 每切换一次 mode，generation+1
    # 旧的 cortex_loop 检测到 generation 不匹配 → 自动终止

    self._mode_transition_event = asyncio.Event()
    # 模式切换信号量: tick 循环检测到需要切换时，设置此 event
    # _handle_mode_transitions() 等待此 event，被唤醒后执行切换

    self._is_reloading = False
    self._mode_initialized = False

    # ====== 第5层: 热重载 ======
    self.hot_reload = hot_reload
    self.check_interval = check_interval  # 默认 60s
    if self.hot_reload:
        self.config_path = self.mode_manager._get_runtime_config_path()
        self.last_modified = os.path.getmtime(config_path)

    run() 方法 — 主事件循环¶

    这是 asyncio.run(runtime.run()) 实际调用的入口：

    async def run(self):
    # ══════════ 阶段 1: 全局启动钩子 ══════════
    await self.mode_config.execute_global_lifecycle_hooks(
        LifecycleHookType.ON_STARTUP, startup_context
    )

    # ══════════ 阶段 2: 初始化第一个 mode ══════════
    await self._initialize_mode(self.mode_manager.current_mode_name)
    #  ↓ 内部:
    #   1. hardware_preflight: 检查硬件状态
    #   2. mode_config.load_components(self.mode_config):
    #        _load_mode_components() → 把所有 _raw_* dict 实例化
    #        → load_input() × N, load_action() × N, load_background() × N, load_llm()
    #   3. self.fuser = Fuser(config)  ← prompt 融合器
    #   4. self.action_orchestrator = ActionOrchestrator(config)
    #   5. self.background_orchestrator = BackgroundOrchestrator(config)
    #   6. self.simulator_orchestrator = SimulatorOrchestrator(config)  # 如有
    self._mode_initialized = True

    # ══════════ 阶段 3: 当前 mode 的启动钩子 ══════════
    await initial_mode_config.execute_lifecycle_hooks(
        LifecycleHookType.ON_STARTUP, startup_context
    )
    # → 你的 welcome config 中: TTS 播放 "你好，我是 Koala..."

    # ══════════ 阶段 4: 启动全部 Orchestrator ══════════
    await self._start_orchestrators()

    # ══════════ 阶段 5: 热重载监视 ══════════
    if self.hot_reload:
        self.config_watcher_task = asyncio.create_task(self._check_config_changes())

    # ══════════ 阶段 6: 无限等待循环 ══════════
    while True:
        # 收集所有还活着的 task
        awaitables = []
        if self.cortex_loop_task and not self.cortex_loop_task.done():
            awaitables.append(self.cortex_loop_task)
        if self.input_listener_task and not self.input_listener_task.done():
            awaitables.append(self.input_listener_task)
        # ... action_task, background_task, simulator_task, mode_transition_task

        # gather 等待所有 task → 任何一个完成或失败就会返回
        await asyncio.gather(*awaitables)
        # 然后重新循环 → 如果 task 因为 CancelledError 结束 → 收集剩余 task 继续等待

    _start_orchestrators() — 六路并发的起点¶

    async def _start_orchestrators(self):
    # 重新允许 tick 间的 sleep (mode 切换时被设为 skip)
    self.sleep_ticker_provider.skip_sleep = False

    # 1. 创建 InputOrchestrator 并启动所有 input 的 listen 循环
    self.input_orchestrator = InputOrchestrator(self.current_config.agent_inputs)
    self.input_listener_task = asyncio.create_task(
        self.input_orchestrator.listen()
    )
    # 内部: 对每个 Sensor → asyncio.create_task(_listen_to_input(sensor))
    #   _listen_to_input → async for event in sensor.listen()
    #     → FuserInput._listen_loop() → yield await self._poll()
    #       → await sensor.raw_to_text(event)

    # 2-4. 启动其他 Orchestrator (都是 asyncio 兼容的 Future)
    if self.simulator_orchestrator:
        self.simulator_task = self.simulator_orchestrator.start()
    if self.action_orchestrator:
        self.action_task = self.action_orchestrator.start()
    if self.background_orchestrator:
        self.background_task = self.background_orchestrator.start()

    # 5. 启动 MCP 服务器
    if self.mcp_orchestrator:
        await self.mcp_orchestrator.start()

    # 6. 启动核心 tick 循环 ← 整个系统的心脏！
    self.cortex_loop_task = asyncio.create_task(self._run_cortex_loop())

    # 7. 启动模式切换处理器
    if not self.mode_transition_task or self.mode_transition_task.done():
        self.mode_transition_task = asyncio.create_task(self._handle_mode_transitions())

    _run_cortex_loop() — 核心 Ticker¶

    async def _run_cortex_loop(self):
    current_mode = self.mode_manager.current_mode_name
    cortex_generation = self._cortex_loop_generation  # ← 捕获当前 generation

    while True:
        # 0. 检查 generation 是否仍然有效
        if not self._is_generation_valid(cortex_generation, "cortex loop"):
            return  # ← mode 已切换，当前 loop 退出

        # 1. 计算休眠时间 (1 / hertz)
        skip = self.sleep_ticker_provider.skip_sleep
        sleep_duration = 1 / self.current_config.hertz  # hertz=2 → 0.5s

        # 2. 休眠 (除非被 skip)
        if not skip and self.current_config:
            await self.sleep_ticker_provider.sleep(sleep_duration)

        # 3. 让出 event loop 控制权
        await asyncio.sleep(0)

        # 4. 执行一个 tick 的全部逻辑
        await self._tick(cortex_generation)

        # 5. 重置 skip 标志
        self.sleep_ticker_provider.skip_sleep = False

    _tick() — 单个 Tick 的完整处理¶

    async def _tick(self, cortex_generation):
    # 1. flush 上一轮完成的 action promises
    finished_promises, _ = await self.action_orchestrator.flush_promises()

    # 2. Fuser.fuse() — 把所有 input 拼成 LLM prompt
    prompt = await self.fuser.fuse(self.current_config.agent_inputs, finished_promises)
    if prompt is None:
        return  # 没有需要处理的内容

    # 3. 检查是否需要模式切换
    last_input = self.io_provider.get_mode_transition_input()
    transition_result = await self.mode_manager.process_tick(last_input)
    if transition_result:
        new_mode, reason = transition_result
        self._pending_mode_transition = new_mode
        self._mode_transition_event.set()  # ← 唤醒 _handle_mode_transitions()
        return  # ← 跳过 LLM 调用，先切换模式

    # 4. action_gate 检查
    if not self._has_fresh_required_input_for_llm():
        return  # ← 没有 fresh Voice → 不调用 LLM

    # 5. voice_planner (可选)
    if await self._run_voice_planner_if_configured(prompt, cortex_generation):
        return  # ← voice planner 已处理

    # 6. 调用 LLM + 执行 action
    await self._execute_cortex_prompt(prompt, cortex_generation)

    _on_mode_transition() — 模式切换回调¶

    async def _on_mode_transition(self, from_mode, to_mode):
    self._is_reloading = True

    # 1. 停止所有旧 mode 的 Orchestrator
    await self._stop_current_orchestrators()
    #  → cancel cortex_loop_task, input_listener_task, action_task... 等全部 task
    #  → cortex_generation+1 → 所有旧 loop 检测到 mismatch → 自行退出
    #  → 等待所有 task CancelledError 完成 (15s 超时)

    # 2. 加载新 mode 的组件
    await self._initialize_mode(to_mode)
    #  → 执行旧 mode ON_EXIT hooks
    #  → 执行全局 ON_EXIT hooks
    #  → load_components(新 mode)
    #  → 创建新的 Fuser, ActionOrchestrator, BackgroundOrchestrator

    # 3. 启动新 mode 的 Orchestrator
    await self._start_orchestrators()
    #  → 新的 input_listener, cortex_loop, background 等 task

    self._is_reloading = False

  │       返回的对象: ModeSystemConfig(

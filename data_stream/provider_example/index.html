    KokoroTTSProvider — TTS 调用链详解¶

    TTS 调用链涉及三层：

    • KokoroTTSProvider — 封装层，单例，提供 add_pending_message() 接口
    • AudioOutputLiveStream — 底层流处理（来自 om1_speech 库），管理 HTTP 调用和 ffplay 进程
    • ffplay 子进程 — ffmpeg 的播放器，实际将 PCM 音频输出到扬声器

    二、KokoroTTSProvider — 薄封装层¶

    @singleton
class KokoroTTSProvider:
    def __init__(self, url, voice_id, model_id, output_format, rate, ...):
        # 创建底层音频流对象
        self._audio_stream = AudioOutputLiveStream(
            url=url,                    # "http://127.0.0.1:8880/v1"
            tts_model=model_id,        # "kokoro"
            tts_voice=voice_id,        # "zm_yunxi"
            response_format=output_format,  # "wav"
            rate=rate,                 # 24000
            enable_tts_interrupt=...,  # 支持 ASR 打断
        )

    def add_pending_message(self, message: Union[str, dict]):
        # 字符串 → dict 包装
        if isinstance(message, str):
            message = self.create_pending_message(message)
            # → {"text": "你好，我是 Koala...",
            #    "voice_id": "zm_yunxi",
            #    "model_id": "kokoro",
            #    "output_format": "wav"}

        # 推入底层队列
        self._audio_stream.add_request(message)

    def start(self):
        self.running = True
        self._audio_stream.start()  # 启动两个 daemon 线程

    def stop(self):
        self.running = False
        self._audio_stream.stop()

    这是一个薄的封装层，核心逻辑都在 AudioOutputLiveStream 中。

    三、AudioOutputLiveStream 初始化¶

    class AudioOutputLiveStream:
    def __init__(self, url, tts_model, tts_voice, ...):
        # 1. 创建 OpenAI 客户端 (连接 Kokoro-FastAPI)
        self.openai_client = openai.OpenAI(
            base_url=self._url,      # "http://127.0.0.1:8880/v1"
            api_key=self._api_key or "no-need-api-key",
        )

        # 2. 初始化 Zenoh (发布/订阅音频状态)
        self.session = open_zenoh_session()
        self.pub = self.session.declare_publisher("robot/status/audio")
        self.session.declare_subscriber("robot/status/audio", self.zenoh_audio_message)
        self.session.declare_subscriber("om/asr/text", self._on_asr_text)  # ASR 打断

        # 3. 线程安全队列
        self._pending_requests: Queue = Queue()

        # 4. ffplay 进程引用
        self._ffplay_proc = None
        self._ffplay_lock = threading.RLock()

    四、start() — 启动两个后台线程¶

    def start(self):
    # 线程1: 核心处理线程 — 消费 TTS 队列，调 API，送音频到 ffplay
    process_thread = threading.Thread(target=self._process_audio)
    process_thread.daemon = True
    process_thread.start()

    # 线程2: 保活线程 — 每 60 秒播放一段静音，防止蓝牙设备休眠
    keepalive_thread = threading.Thread(target=self._keepalive_worker)
    keepalive_thread.daemon = True
    keepalive_thread.start()

    五、_process_audio() — 核心 TTS 处理循环¶

    def _process_audio(self):
    while self.running:
        tts_request = None
        try:
            # ═══ 第1步: 从队列取出请求 (阻塞等待) ═══
            tts_request = self._pending_requests.get()
            if tts_request is None:   # None 是停止信号
                break

            request_id = tts_request.get("request_id", "")

            # ═══ 第2步: 确保 ffplay 进程在运行 ═══
            if not self._initialize_ffplay():
                continue

            # ═══ 第3步: 播放一小段前置静音 (给蓝牙设备唤醒时间) ═══
            self._stream_audio_chunk(self._create_silence_audio(10))

            # ═══ 第4步: 通知回调 + 发布 Zenoh 状态 ═══
            self._tts_callback(True)
            self._update_audio_status(
                AudioStatus.STATUS_SPEAKER.ACTIVE.value, request_id
            )

            # ═══ 第5步: 调用 Kokoro TTS API (OpenAI 兼容，流式) ═══
            with self.openai_client.audio.speech.with_streaming_response.create(
                model=self._tts_model,        # "kokoro"
                voice=self._tts_voice,        # "zm_yunxi"
                response_format=self._response_format,  # "wav"
                input=tts_request["text"],    # "你好，我是 Koala..."
                extra_body=self._extra_body,
            ) as response:
                # ═══ 第6步: 逐个 1024 字节 chunk 写入 ffplay stdin ═══
                for chunk in response.iter_bytes(chunk_size=1024):
                    if not self.running:
                        break
                    self._stream_audio_chunk(chunk)

            # ═══ 第7步: 关闭 stdin，等待 ffplay 播完 ═══
            self._finish_audio_playback()

            # ═══ 第8步: 通知回调 + 发布 Zenoh 状态 ═══
            self._tts_callback(False)
            self._update_audio_status(
                AudioStatus.STATUS_SPEAKER.READY.value, request_id
            )

        except Exception as e:
            logger.error(f"Error processing audio: {e}")
            self._tts_callback(False)
            continue

    关键步骤详解¶

    第2步：_initialize_ffplay() — 启动 ffplay 子进程¶

    def _initialize_ffplay(self) -> bool:
    with self._ffplay_lock:
        # 如果 ffplay 已经在运行，直接返回
        if self._ffplay_proc and self._ffplay_proc.poll() is None:
            return True

        # 检查 ffplay 是否安装
        if not shutil.which("ffplay"):
            return False

        # 构建命令行参数
        args = ["ffplay"]
        if "pcm" in self._response_format:
            args.extend(["-f", "s16le", "-ar", str(self._rate)])
            # -f s16le: signed 16-bit little-endian PCM
            # -ar 24000: 采样率 24kHz
        args.extend(["-nodisp", "-autoexit", "-"])
            # -nodisp:  不显示视频窗口
            # -autoexit: 播完自动退出
            # -:         从 stdin 读取

        # 启动子进程，stdin 管道
        self._ffplay_proc = subprocess.Popen(
            args=args,
            stdin=subprocess.PIPE,      # 通过管道写入音频数据
            stdout=subprocess.DEVNULL,   # 丢弃 stdout
            stderr=subprocess.DEVNULL,   # 丢弃 stderr
            bufsize=0,                   # 无缓冲（实时流）
        )
        return True

    第5步：TTS API 调用（实际 HTTP 请求）¶

    self.openai_client.audio.speech.with_streaming_response.create(
    model="kokoro",
    voice="zm_yunxi",
    response_format="wav",
    input="你好，我是 Koala...",
)

    底层 HTTP 请求：

    POST http://127.0.0.1:8880/v1/audio/speech
Content-Type: application/json

{
    "model": "kokoro",
    "voice": "zm_yunxi",
    "response_format": "wav",
    "input": "你好，我是 Koala。你可以告诉我你的名字，或者直接说需要我帮什么。"
}

→ 响应: streaming audio/wav 字节流

    关键点：URL 是 http://127.0.0.1:8880/v1，这是本地运行的 Kokoro-FastAPI Docker 服务，它实现了 OpenAI-compatible 的 /v1/audio/speech 接口。所以可以直接用 OpenAI Python SDK 调用，不需要特殊的 TTS SDK。

    with_streaming_response.create() 是 OpenAI SDK 的流式模式：HTTP 响应是 Transfer-Encoding: chunked，SDK 边接收边 yield 字节块，实现边合成边播放。

    第6步：_stream_audio_chunk() — 实时推流¶

    def _stream_audio_chunk(self, chunk: bytes):
    with self._ffplay_lock:
        if self._ffplay_proc and self._ffplay_proc.poll() is None:
            if self._ffplay_proc.stdin:
                self._ffplay_proc.stdin.write(chunk)   # 写入 ffplay stdin
                self._ffplay_proc.stdin.flush()         # 立即刷新 (bufsize=0)

    每个 1024 字节音频 chunk 到达后立刻写入 ffplay 管道。ffplay 从 stdin 读取、解码 wav/pcm、通过 PulseAudio/ALSA 输出到物理扬声器。

    第7步：_finish_audio_playback() — 等待播放完成¶

    def _finish_audio_playback(self):
    with self._ffplay_lock:
        if self._ffplay_proc and self._ffplay_proc.poll() is None:
            # 关闭 stdin → ffplay 收到 EOF
            self._ffplay_proc.stdin.close()
            # 等待 ffplay 播完退出 (最多 10 秒)
            self._ffplay_proc.wait(timeout=10)
            self._ffplay_proc = None
            self._ffplay_initialized = False  # 下次会重建

    stdin 关闭后，ffplay 收到 EOF，配合 -autoexit 自动退出。下一次 TTS 请求时重新创建 ffplay 进程。

    _keepalive_worker() — 蓝牙保活¶

    def _keepalive_worker(self):
    while self.running:
        current_time = time.time()
        if current_time - self._last_audio_time >= 60:
            self._write_audio_bytes(self._silence_audio)  # 写一段静音
            self._last_audio_time = current_time
        time.sleep(10)

    每 60 秒播放 50ms 静音。原因是某些蓝牙设备在长时间无音频时会进入休眠模式，恢复时有明显延迟（0.5-2s）。保活静音让蓝牙始终保持激活。

    六、ASR 打断机制¶

    # AudioOutputLiveStream 初始化时订阅了 ASR 文本 topic
self.session.declare_subscriber("om/asr/text", self._on_asr_text)

def _on_asr_text(self, data):
    if (self._enable_tts_interrupt
        and self.audio_status.status_speaker == AudioStatus.STATUS_SPEAKER.ACTIVE.value):
        # 清空 pending 队列
        while not self._pending_requests.empty():
            self._pending_requests.get_nowait()
        # 杀掉 ffplay (中止当前播放)
        self._cleanup_ffplay()

    当用户说话打断机器人时： 1. ASR 检测到语音 → 发布 om/asr/text 2. _on_asr_text 被触发 → 检查 enable_tts_interrupt=true 且 SPEAKER=ACTIVE 3. 清空后续待播放消息 4. 立即 terminate() ffplay 进程，停止当前播放

    七、两种 Speak Action 分发路径¶

    # src/actions/speak/connector/kokoro_tts.py → SpeakKokoroTTSConnector.connect()

async def connect(self, output_interface):
    pending_message = self.tts.create_pending_message(output_interface.action)

    # 路径 A: 通过 Zenoh 发布 (远程播放 — Go2 机器人扬声器)
    if self.audio_pub:
        state = AudioStatus(
            status_speaker=AudioStatus.STATUS_SPEAKER.ACTIVE.value,
            sentence_to_speak=String(json.dumps(pending_message)),
        )
        self.audio_pub.put(state.serialize())
        return

    # 路径 B: 直接本地播放 (开发调试)
    self.tts.add_pending_message(pending_message)
    路径 触发条件 流程 场景
    A: Zenoh 发布 audio_pub 存在 SpeakAction → AudioStatus → Zenoh topic robot/status/audio → Go2 上的 AudioOutputLiveStream 收到 → 调 TTS API → ffplay 在机器人上播放 机器人扬声器播放
    B: 直接播放 audio_pub 为 None (Zenoh 未初始化) SpeakAction → KokoroTTSProvider.add_pending_message() → AudioOutputLiveStream._process_audio() → 本地 ffplay 开发调试

    函数签名与位置¶

    # src/runtime/cortex.py, 行 1537

async def _execute_cortex_prompt(self, prompt: str, cortex_generation: int) -> None:

    输入：Fuser 融合好的完整 prompt 文本（已经过子任务替换、Voice 注入） 输出：无返回值（副作用：调用 LLM → 派发 actions → 驱动硬件）

    五阶段执行流程¶

    _execute_cortex_prompt(prompt, cortex_generation)
    │
    ├── ① 调用 LLM，流式获取结果
    │     async for output in cortex_llm.ask_stream(prompt):
    │         拿到 CortexOutputModel(actions=[Action(...), Action(...)])
    │
    ├── ② MCP Tool Calling Loop（如果 LLM 输出包含 mcp_ 工具调用）

    源码对应¶

    async def _execute_cortex_prompt(self, prompt: str, cortex_generation: int) -> None:
    output = None
    try:
        # ① 流式调用 LLM
        async for output in self.current_config.cortex_llm.ask_stream(prompt):
            if output is None:
                return

            # ② MCP Tool Calling Loop
            if self.mcp_orchestrator:
                succeeded_calls = set()
                for round_idx in range(self.mcp_orchestrator.max_rounds):
                    om1_actions = self.mcp_orchestrator.extract_om1_actions(output.actions)
                    results, mcp_actions = await self.mcp_orchestrator.execute_mcp_actions(
                        output.actions, succeeded_calls
                    )
                    if results is None:
                        break
                    if om1_actions:
                        await self.action_orchestrator.promise(om1_actions)
                    # 回填结果 → 重新调 LLM
                    recall_prompt = self.mcp_orchestrator.build_result_prompt(original_prompt, results)
                    async for stream_output in self.current_config.cortex_llm.ask_stream(recall_prompt):
                        ...
                # 最终 output.actions = 纯 OM1 actions
                output.actions = self.mcp_orchestrator.extract_om1_actions(output.actions)

            # ③ 修复遗漏动作
            output.actions = repair_speak_only_voice_intents(
                prompt, output.actions,
                available_action_types=_available_action_types_from_config(self.current_config),
            )

            # ④ 通知 Simulator
            if self.simulator_orchestrator:
                await self.simulator_orchestrator.promise(output.actions)

            # ④ 标记子任务步骤已开始
            self._mark_voice_task_started_for_action_dispatch(output.actions)

            # ⑤ ★ 派发到执行器
            await self.action_orchestrator.promise(output.actions)
    except asyncio.CancelledError:
        raise

    阶段 ⑤ 详解：ActionOrchestrator.promise()¶

    这是最关键的环节。ActionOrchestrator 位于 src/actions/orchestrator.py，它接收 LLM 输出的原始 Action 对象，逐一找到对应的硬件执行器（Connector），做参数校验和类型转换，最终调用 connector.connect()。

    步骤 1：遍历 actions¶

    # cortex.py 调用:
await self.action_orchestrator.promise([
    Action(type="speak", value="好的，我去仓库拿饮料"),
    Action(type="navigate_location", value="仓库"),
])

    # actions/orchestrator.py:
async def promise(self, actions: list[Action]) -> None:
    # 默认 concurrent 模式：所有 action 并发派发
    for action in actions:
        normalized = self._normalize_action(action)
        agent_action = self._get_agent_action(normalized)
        asyncio.create_task(self._promise_action(agent_action, normalized))

    步骤 2：标准化动作名 (normalize)¶

    某些 LLM 可能输出的口语化动作名被转换为标准名：

    def _normalize_action(self, action: Action) -> Action:
    at = action.type.lower()
    # "stand still"   → Action(type="move", value="stand still")
    # "turn left"     → Action(type="move", value="turn left")
    # "move forwards" → Action(type="move", value="move forwards")
    # "navigate_location" → 不变
    return action

    步骤 3：Action Gate 检查¶

    这是安全闸门——检查当前动作是否允许执行，防止 LLM 在"没有新鲜输入"的情况下乱发指令：

    def _is_action_allowed_by_gate(self, action: Action) -> bool:
    gate = self._config.action_gate
    # gate 配置示例:
    # { gated_actions: ["speak", "navigate_location", "go2_posture", "hello"],
    #   require_fresh_input_key: "Voice",
    #   voice_carryover_ticks: 0 }

    gated_actions = {str(a).lower() for a in gate.get("gated_actions", [])}
    if action.type.lower() not in gated_actions:
        return True  # 不在管制名单中 → 放行

    # 在管制名单中 → 检查 Voice 是否新鲜
    voice = self.io_provider.get_input("Voice")
    if voice is None or not voice.input.strip():
        return False      # Voice 不存在或为空 → 阻止
    if voice.tick != self.io_provider.tick_counter:
        return False      # Voice 是旧 tick 的残留 → 阻止
    return True

    如果一个动作被 Gate 阻止，它不会被派发给 Connector，直接被丢弃。

    步骤 4：匹配 AgentAction¶

    用 LLM 输出的 type 字段去配置中查找对应的 AgentAction 对象：

    def _get_agent_action(self, action: Action) -> Optional[AgentAction]:
    return next(
        (m for m in self._config.agent_actions
         if m.llm_label == action.type.lower()),
        None,
    )
    # action.type = "navigate_location"
    # → 在 agent_actions 列表中找到 llm_label="navigate_location" 的配置项
    # → 返回 AgentAction(name="navigate_location", connector=AutonomyMid360NavConnector, ...)

    配置文件中的对应关系：

    agent_actions: [
  {
    name: "navigate_location",            // 内部名称
    llm_label: "navigate_location",       // ← LLM 输出的 type 必须匹配这个
    connector: "autonomy_mid360_nav",     // 使用哪个 Connector 类
    interface: "NavigateLocationInput",   // 输入参数 Pydantic 模型
  },
  {
    name: "speak",
    llm_label: "speak",
    connector: "qwen_tts",
    interface: "SpeakInput",
  },
]

    步骤 5：参数解析 & 类型转换¶

    这是最精妙的部分——将 LLM 输出的纯字符串 value 转换为 Connector 期望的强类型 Pydantic 对象：

    async def _promise_action(self, agent_action: AgentAction, action: Action):
    # ─── 5a. 解析 action.value ───
    # value="仓库"
    #   → parsed_value = {"action": "仓库"}
    # value='{"target_name":"bottle","top_k":50}'
    #   → parsed_value = {"target_name": "bottle", "top_k": 50}

    try:
        parsed_value = json.loads(action.value)
        if isinstance(parsed_value, dict):
            input_params = parsed_value
        else:
            input_params = {"action": action.value}
    except (json.JSONDecodeError, TypeError):
        input_params = {"action": action.value}

    # ─── 5b. 获取 Connector 期望的 Pydantic 类型 ───
    # agent_action.interface = Interface[NavigateLocationInput, ...]
    input_type = get_type_hints(agent_action.interface)["input"]
    # → NavigateLocationInput(action: str)

    # ─── 5c. 类型转换（string → int/float/bool/Enum） ───
    type_hints = get_type_hints(input_type)
    converted_params = {}
    for key, value in input_params.items():
        expected_type = type_hints[key]
        if is_enum(expected_type):
            converted_params[key] = expected_type(value)
            # "grasp" → GraspAction.GRASP
        elif expected_type is float:
            converted_params[key] = float(value)
        elif expected_type is int:
            converted_params[key] = int(value)
        elif expected_type is bool:
            converted_params[key] = value.lower() in ("true", "1", "yes")
        else:
            converted_params[key] = value

    # ─── 5d. 构造 Pydantic 输入对象 ───
    input_interface = input_type(**converted_params)
    # → NavigateLocationInput(action="仓库")

    # ─── 5e. ★ 调用 Connector! ───
    await agent_action.connector.connect(input_interface)

    步骤 6：Connector.connect() — 硬件交互¶

    每种动作有不同的 Connector 实现，但接口统一：

    动作 Connector connect() 实际操作
    navigate_location AutonomyMid360NavConnector 查 JSON 文件得到目标位姿 → 调 publish_goal.sh → 发布 ROS 2 /goal_point
    speak SpeakQwenTTSConnector 调 Qwen TTS API → 播放语音
    grasp_object ARXX5GraspZenohConnector Zenoh publish arx_x5/grasp/request → 后台 Executor 收到 → 控制机械臂
    hello ARXX5HelloZenohConnector Zenoh publish 挥手动作指令
    go2_posture Go2PostureConnector Unitree SportClient → StandUp / StandDown

    导航 Connector 的例子：

    class AutonomyMid360NavConnector(ActionConnector):
    async def connect(self, output_interface: NavigateLocationInput):
        target = output_interface.action   # "仓库"
        # 1. 查 JSON: {"仓库": [[x, y, z, qx, qy, qz, qw], ...]}
        pose = self._lookup_pose(target)
        # 2. 调 shell 脚本发布 ROS 2 /goal_point
        subprocess.run([
            "bash", "publish_goal.sh",
            str(pose[0]), str(pose[1]), str(pose[2]),
            str(pose[3]), str(pose[4]), str(pose[5]), str(pose[6]),
        ])

    抓取 Connector 的例子：

    class ARXX5GraspZenohConnector(ActionConnector):
    async def connect(self, output_interface: ARXX5GraspInput):
        payload = {
            "action": "grasp",
            "target_name": output_interface.target_name,  # "bottle"
            "top_k": output_interface.top_k,              # 50
        }
        # Zenoh publish → 后台 Executor 收到 → 控制机械臂
        self.session.put("arx_x5/grasp/request", ZBytes(json.dumps(payload)))

    三种执行模式¶

    模式 行为 配置字段
    concurrent（默认） 所有 action 同时派发，create_task 后不等待，谁先完成谁先返回 不配置或 action_execution_mode: "concurrent"
    sequential 逐个执行，await 第 1 个 action 完成后才派发第 2 个 action_execution_mode: "sequential"
    dependencies 按依赖图执行：B 等 A 的 completed_actions Event 被 set 后才执行 action_execution_mode: "dependencies" + action_dependencies: {speak: [], navigate: ["speak"]}

    关键设计要点¶

    1. LLM 输出是松散的 Action(type, value) — LLM 不需要知道参数的类型，只需输出字符串。类型转换和校验在 _promise_action() 中自动完成。

    2. llm_label 是 LLM 和配置之间的唯一桥梁 — LLM 输出的 type 必须是配置中某个 AgentAction.llm_label 的值。匹配失败 → 动作被丢弃（只打 warning 日志）。

    3. Action Gate 做了双重检查 — _tick() 入口处检查 + _promise_action() 派发时再检查，双重保险。

    4. Connector 接口统一 — 所有 Connector 都是 async connect(input: PydanticModel)，不管底层是 Zenoh、ROS 2、subprocess 还是 HTTP API。

    5. Connector 只发送指令，不采集数据 — 传感器数据由 Input + Provider 负责，Background 负责持续驱动。Connector 只管执行。

    6. 新增动作只需两步 — (a) 在 JSON5 中加一条 agent_actions 配置，(b) 写一个 Connector 类实现 connect()。不需要改 LLM 层或 Cortex 的代码。

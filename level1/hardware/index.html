    Hardware Preflight 硬件自检机制详解¶

    代码位置：src/runtime/hardware_preflight.py（533 行）

    调用入口：src/runtime/cortex.py:_initialize_mode()，在 load_components() 之前执行——自检操作的是 _raw_* 原始 dict，在组件实例化前就完成过滤。

    一、调用入口¶

    cortex.py:287：

    async def _initialize_mode(self, mode_name):
    mode_config = self.mode_config.modes[mode_name]

    # ═══ 第1步: 硬件自检 ← 在 load_components 之前! ═══
    preflight_report = apply_hardware_preflight(mode_config)

    # ═══ 第2步: 加载组件 (此时 _raw_* 已被自检过滤) ═══
    mode_config.load_components(self.mode_config)

    # ═══ 第3步: 将自检报告注入 LLM 上下文 ═══
    if preflight_report is not None:
        _store_hardware_preflight_report(self.io_provider, preflight_report)

    # ═══ 第4步: TTS 播报自检完成 ═══
    _announce_hardware_preflight_completion(mode_config)

    为什么必须在 load_components() 之前？ 自检发现某硬件不可用后，直接把它依赖的 input/background/action 从 _raw_* 列表中删除，这些组件根本不会被实例化。

    二、apply_hardware_preflight() 四步流程¶

    def apply_hardware_preflight(mode_config, probe=None):
    # 0. 恢复原始 _raw_* 配置 (如果之前被 hot-reload 修改过)
    _restore_original_raw_config(mode_config)

    report = HardwarePreflightReport()
    preflight_config = getattr(mode_config, "hardware_preflight", None)
    if enabled == False:
        return report  # ← 配置关闭了自检

    active_probe = probe or HardwarePreflightProbe()

    # ═══ 第1步: 探测配置中声明的 capabilities ═══
    configured_capabilities = _probe_configured_capabilities(
        preflight_config, active_probe, report
    )

    # ═══ 第2步: 如果没有显式声明 capabilities，从 D435 background 推断 ═══
    d435_capabilities = _d435_capabilities_from_report(report)
    if not configured_capabilities:
        d435_capabilities = _probe_d435_backgrounds(
            mode_config._raw_backgrounds, active_probe, report
        )

    # ═══ 第3步: 探测 ARX5 机械臂 (如果配置使用了它) ═══
    arx_required = _mode_uses_arx_motion(mode_config)
    if arx_required and "arm:arx_x5:motion" not in report.available_capabilities:
        available, detail = active_probe.has_fresh_arx_state(
            topic="arx_x5/state", timeout_seconds=5.0,
            max_age_seconds=2.0, required_fields=["joint_pos"],
        )
        report.available_capabilities["arm:arx_x5:motion"] = available
        report.checks.append(HardwareCheck("ARX X5 motion", available, detail))

    # ═══ 第4步: 根据探测结果，从 _raw_* 中移除不满足条件的组件 ═══
    mode_config._raw_inputs = [
        inp for inp in mode_config._raw_inputs
        if _input_enabled(inp, d435_capabilities, report)
    ]
    mode_config._raw_backgrounds = [
        bg for bg in mode_config._raw_backgrounds
        if _background_enabled(bg, d435_capabilities, arx_available, report)
    ]
    mode_config._raw_actions = [
        act for act in mode_config._raw_actions
        if _action_enabled(act, d435_capabilities, grasp_camera_id, arx_available, report)
    ]

    _log_report(report)
    return report

    三、两种硬件探针 (Probe) 的具体实现¶

    配置中的 probe 字段决定探针类型。以 arm_x5_grasping.json5 为例：

    capabilities: {
    "camera:d435:default": {
        probe: "realsense",                    // ← 探针类型 1
        serial_number: "346522071646",
    },
    "camera:d435:camera_2": {
        probe: "realsense",
        serial_number: "346522075383",
    },
    "arm:arx_x5:motion": {
        probe: "zenoh_state",                  // ← 探针类型 2
        topic: "arx_x5/state",
        required_fields: ["joint_pos"],
        timeout_seconds: 5.0,
        max_age_seconds: 2.0,
    },
}

    探针1: realsense — 检查 D435 相机¶

    class HardwarePreflightProbe:
    def has_realsense_device(self, serial_number=None) -> tuple[bool, str]:
        import pyrealsense2 as rs

        # 查询 USB 总线上的所有 RealSense 设备
        devices = list(rs.context().query_devices())

        if not devices:
            return False, "no RealSense devices found"

        # 指定序列号 → 精确匹配
        if serial_number:
            for device in devices:
                serial = device.get_info(rs.camera_info.serial_number)
                if serial == serial_number:
                    return True, f"RealSense serial {serial_number} found"
            return False, f"RealSense serial {serial_number} not found"

        # 未指定序列号 → 只要有设备就通过
        return True, f"{len(devices)} RealSense device(s) found"

    调用 rs.context().query_devices() 走 USB 协议查询所有连接的 RealSense 相机，按序列号精确匹配。

    探针2: zenoh_state — 检查 ARX5 机械臂¶

    def has_fresh_arx_state(self, *, topic, timeout_seconds,
                         max_age_seconds, required_fields=None):
    state_event = threading.Event()
    fields = required_fields or ["joint_pos"]

    def _on_state(sample):
        payload = json.loads(sample.payload.to_bytes().decode("utf-8"))
        # 检查必须字段
        missing = [f for f in fields if f not in payload]
        if missing:
            return
        # 检查时间戳新鲜度 (< max_age_seconds)
        fresh, _ = _arx_state_freshness_detail(payload, max_age_seconds)
        if not fresh:
            return
        state_event.set()  # ← 通过! 唤醒等待线程

    # 打开 Zenoh 会话，订阅 topic
    session = open_zenoh_session()
    subscriber = session.declare_subscriber(topic, _on_state)

    # 阻塞等待，直到收到新鲜数据或超时
    if state_event.wait(timeout=5.0):
        return True, "fresh state received with fields: joint_pos"
    else:
        return False, "no fresh ARX X5 state within 5.0s"
    1. 打开 Zenoh 客户端
    2. 订阅 arx_x5/state
    3. 阻塞等待最多 5 秒，看是否有包含 joint_pos 且时间戳在 2 秒内的消息
    4. 收到 → 机械臂在线；超时 → 不可用

    四、组件禁用规则¶

    requires_capabilities 机制¶

    配置中组件可以声明 requires_capabilities：

    // 例: grasp action 需要相机和机械臂同时在线
{
    name: "arm_arx_x5_yolograsp",
    llm_label: "grasp_object",
    requires_capabilities: ["camera:d435:default", "arm:arx_x5:motion"],
}

    自检后框架遍历每个组件的 requires_capabilities，如果有任何一个 capability 不可用，该组件从 _raw_* 列表中直接删除：

    def _requirements_satisfied(entry, *, required_capabilities, report, disabled):
    missing = [
        cap for cap in required_capabilities
        if not report.available_capabilities.get(cap, False)
    ]
    if not missing:
        return True  # 所有依赖都满足

    disabled.append(component_name)
    return False  # 移除该组件

    Input 禁用规则¶

    def _input_enabled(input_config, d435_capabilities, report):
    # 方式1: 显式 requires_capabilities
    required = _requires_capabilities(input_config)
    if required is not None:
        return _requirements_satisfied(...)

    # 方式2: D435RealSenseInput 无显式声明时，用 camera_id 推断
    if input_config.get("type") == "D435RealSenseInput":
        camera_id = config.get("camera_id", "default")
        if d435_capabilities.get(camera_id, False):
            return True
        report.disabled_inputs.append(f"D435RealSenseInput(camera_id={camera_id})")
        return False

    return True

    Background 禁用规则¶

    def _background_enabled(bg_config, d435_capabilities, arx_available, report):
    # 方式1: 显式 requires_capabilities
    required = _requires_capabilities(bg_config)
    if required is not None:
        return _requirements_satisfied(...)

    # 方式2: D435 background — 相机不可用则禁用
    if bg_type == "D435":
        camera_id = config.get("camera_id", "default")
        if d435_capabilities.get(camera_id, False):
            return True
        report.disabled_backgrounds.append(f"D435(camera_id={camera_id})")
        return False

    # 方式3: ARX5 抓取执行器 — 机械臂不可用则禁用
    if bg_type == "ARXX5YoloGraspExecutor" and not arx_available:
        report.disabled_backgrounds.append("ARXX5YoloGraspExecutor")
        return False

    Action 禁用规则¶

    def _action_enabled(action_config, d435_capabilities, grasp_camera_id,
                    arx_available, report):
    label = action_config.get("llm_label", "").lower()

    # 方式1: 显式 requires_capabilities
    ...

    # 方式2: 特殊 action 的硬编码规则
    if label == "grasp_object":
        if arx_available and d435_capabilities.get(grasp_camera_id, False):
            return True
        report.disabled_actions.append("grasp_object")
        return False

    if label in {"arm_x5_reset", "arm_x5_joint_nudge", "hello"}:
        if not arx_available:
            report.disabled_actions.append(label)
            return False

    六、hardware_preflight 的三种配置方式¶

    配置方式 示例 适用场景
    显式 capabilities capabilities: {"camera:d435:default": {probe:"realsense", ...}} 精确控制每个硬件的探测方式
    D435 background 推断 无 capabilities 字段，直接从 backgrounds[type=D435] 检测 简单场景，只检查相机
    禁用 hardware_preflight: {enabled: false} 不需要自检（如纯对话模式）

    八、设计要点总结¶

    设计决策 原因
    在 load_components() 之前执行 不可用硬件对应的组件从根本上不创建，不是"创建后再禁用"
    原始配置深拷贝 + 恢复 hot-reload 时 repeat preflight 不会丢失之前被移除的组件
    requires_capabilities 声明式依赖 新增组件只需声明需要什么硬件，不必修改自检逻辑
    探针可插拔 HardwarePreflightProbe 可被 mock，单元测试不依赖真实硬件
    报告注入 LLM prompt Model 知道自己什么能做、什么不能做，不会尝试调用已被禁用的 action
    TTS 播报 tts_on_complete 让用户知道自检完成，system_prompt_base 告诉 LLM 如何处理硬件异常的询问
